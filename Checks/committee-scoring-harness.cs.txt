class FixtureAuthentication(IOptionsMonitor<AuthenticationSchemeOptions> options,ILoggerFactory logger,UrlEncoder encoder,FixtureActors actors):AuthenticationHandler<AuthenticationSchemeOptions>(options,logger,encoder){
 protected override Task<AuthenticateResult> HandleAuthenticateAsync(){var token=Request.Headers["X-Scoring-Test"].ToString();var key=Request.Headers["X-Member"].ToString();if(Request.Cookies.TryGetValue("CommitteeScoringFixture",out var cookie)&&cookie.StartsWith(actors.Token+".",StringComparison.Ordinal)){token=actors.Token;key=cookie[(actors.Token.Length+1)..];}if(token!=actors.Token||!actors.Users.TryGetValue(key,out var user))return Task.FromResult(AuthenticateResult.NoResult());var role=key=="officer"?RoleNames.SourcingOfficer:key=="manager"?RoleNames.SourcingManager:RoleNames.EvaluationCommittee;var claims=new[]{new Claim(ClaimTypes.NameIdentifier,user.Id),new Claim(ClaimTypes.Name,user.UserName!),new Claim(ClaimTypes.Role,role)};return Task.FromResult(AuthenticateResult.Success(new AuthenticationTicket(new ClaimsPrincipal(new ClaimsIdentity(claims,"ScoringFixture")),"ScoringFixture")));}
}
class NoAi:HttpMessageHandler{
 public static int Calls; public static bool AllowFinal;
 protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage request,CancellationToken ct){
  Interlocked.Increment(ref Calls);
  if(!AllowFinal)throw new Exception("MVC scoring tests must not call a real AI service.");
  using var input=System.Text.Json.JsonDocument.Parse(await request.Content!.ReadAsStringAsync(ct));
  // OpenAI-compatible chat request: the schema is in response_format.
  var props=input.RootElement.GetProperty("response_format").GetProperty("json_schema").GetProperty("schema").GetProperty("properties");
  var chosen=props.GetProperty("selectedBidId").GetProperty("enum")[0].GetInt32();
  var alternative=props.GetProperty("comparison").GetProperty("items").GetProperty("properties").GetProperty("bidId").GetProperty("enum")[0].GetInt32();
  var report=new {selectedBidId=chosen,whyChosen="The selected offer matches the requested sterile packaging and quantities, with documented delivery terms for the pharmacy.",comparison=new[]{new{bidId=alternative,strengths="The alternative also documents sterile packaging and complete quantities for the requested pharmacy supplies.",tradeoffs="The alternative has a higher total cost and a longer stated delivery timeframe than the selected offer."}},observations="No non-compliance is recorded. Supplier packaging and traceability statements remain subject to verification before delivery.",conclusion="Recommend the committee-selected offer based on the documented packaging, quantity and delivery terms for these pharmacy supplies."};
  return new HttpResponseMessage(HttpStatusCode.OK){Content=new StringContent(System.Text.Json.JsonSerializer.Serialize(new{choices=new[]{new{message=new{content=System.Text.Json.JsonSerializer.Serialize(report)}}}}),System.Text.Encoding.UTF8,"application/json")};
 }
}
