class NoAi:HttpMessageHandler{
 public static int Calls; public static bool AllowFinal; public static bool InvalidReferral;
 protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage request,CancellationToken ct){
  Interlocked.Increment(ref Calls);
  if(!AllowFinal)throw new Exception("MVC scoring tests must not call a real AI service.");
  using var input=System.Text.Json.JsonDocument.Parse(await request.Content!.ReadAsStringAsync(ct));
  var props=input.RootElement.GetProperty("format").GetProperty("properties");
  object report;
  if(props.TryGetProperty("summary",out _)){
   using var evidence=System.Text.Json.JsonDocument.Parse(input.RootElement.GetProperty("prompt").GetString()!);
   if(evidence.RootElement.TryGetProperty("committeeSummary",out _)||evidence.RootElement.TryGetProperty("officerSummary",out _)||evidence.RootElement.GetRawText().Contains("Levant Medicine Supply"))throw new Exception("Manager AI received alternative supplier evidence.");
   var purchase=evidence.RootElement.GetProperty("purchaseRequest");
   if(!purchase.GetProperty("reason").GetString()!.Contains("Replenish Tiryaq")||purchase.GetProperty("items")[0].GetProperty("itemName").GetString()!="Sterile gauze swabs"||purchase.GetProperty("items")[0].GetProperty("quantity").GetDecimal()!=120||!evidence.RootElement.GetProperty("offer").TryGetProperty("items",out _))throw new Exception("Manager AI lost purchase requirements or winning offer line items.");
   var supplier=evidence.RootElement.GetProperty("offer").GetProperty("supplier").GetString();
   report=new{summary=$"{supplier} is recommended on the recorded committee score. Final committee review is requested."+(InvalidReferral?" Levant Medicine Supply is an alternative.":"")};
  }else if(props.TryGetProperty("winnerBidId",out var winnerSchema)){
   var winner=winnerSchema.GetProperty("enum")[0].GetInt32();
   using var evidence=System.Text.Json.JsonDocument.Parse(input.RootElement.GetProperty("prompt").GetString()!.Split("\nCorrect the previous response:")[0]);
   var ids=evidence.RootElement.GetProperty("topOffers").EnumerateArray().Select(o=>o.GetProperty("bidId").GetInt32()).ToArray();
   report=new{winnerBidId=winner,offers=ids.Select(id=>new{bidId=id,assessment="The supplier states that the proposed packaging and quantities follow the request. Committee marks establish its recorded position in the review.",caveats="Verify the supplier documentation against the requested specification before purchase; screening is not a guarantee of future performance."}).ToArray()};
  }else{
   var chosen=props.GetProperty("selectedBidId").GetProperty("enum")[0].GetInt32();
   var alternative=props.GetProperty("comparison").GetProperty("items").GetProperty("properties").GetProperty("bidId").GetProperty("enum")[0].GetInt32();
   report=new {selectedBidId=chosen,whyChosen="The selected offer matches the requested sterile packaging and quantities, with documented delivery terms for the pharmacy.",comparison=new[]{new{bidId=alternative,strengths="The alternative also documents sterile packaging and complete quantities for the requested pharmacy supplies.",tradeoffs="The alternative has a higher total cost and a longer stated delivery timeframe than the selected offer."}},observations="No non-compliance is recorded. Supplier packaging and traceability statements remain subject to verification before delivery.",conclusion="Recommend the committee-selected offer based on the documented packaging, quantity and delivery terms for these pharmacy supplies."};
  }
  return new HttpResponseMessage(HttpStatusCode.OK){Content=new StringContent(System.Text.Json.JsonSerializer.Serialize(new{done=true,done_reason="stop",response=System.Text.Json.JsonSerializer.Serialize(report)}),System.Text.Encoding.UTF8,"application/json")};
 }
