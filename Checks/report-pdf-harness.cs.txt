record QaActors(Dictionary<string,(ApplicationUser User,string Role)> Actors,string Token);
class QaAuthentication(IOptionsMonitor<AuthenticationSchemeOptions> options,ILoggerFactory logger,UrlEncoder encoder,QaActors actors)
    : AuthenticationHandler<AuthenticationSchemeOptions>(options,logger,encoder)
{
    protected override Task<AuthenticateResult> HandleAuthenticateAsync()
    {
        if(Request.Headers["X-Report-QA"]!=actors.Token || !actors.Actors.TryGetValue(Request.Headers["X-Report-Actor"].ToString(),out var actor))
            return Task.FromResult(AuthenticateResult.NoResult());
        var claims=new[]{new Claim(ClaimTypes.NameIdentifier,actor.User.Id),new Claim(ClaimTypes.Name,actor.User.UserName!),new Claim(ClaimTypes.Role,actor.Role)};
        return Task.FromResult(AuthenticateResult.Success(new AuthenticationTicket(new ClaimsPrincipal(new ClaimsIdentity(claims,"ReportQA")),"ReportQA")));
    }
}
class NoAiCalls : HttpMessageHandler
{
    public static int Calls;
    protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage request,CancellationToken ct)
    { Interlocked.Increment(ref Calls); throw new InvalidOperationException("PDF tests must never call AI."); }
}
class FixedClock : TimeProvider { public override DateTimeOffset GetUtcNow()=>new(2026,11,14,12,34,56,TimeSpan.Zero); }
class QaWatermark(IHttpContextAccessor accessor):IPdfWatermarkService
{
    public Task<byte[]> ApplyAsync(Stream source,ClaimsPrincipal user,DateTimeOffset at,CancellationToken ct=default)
    {
        if(accessor.HttpContext is {} context && context.Request.Headers["X-Report-Fail-Watermark"].ToString()=="1")throw new InvalidDataException("Synthetic watermark failure");
        return new PdfWatermarkService().ApplyAsync(source,user,at,ct);
    }
}
